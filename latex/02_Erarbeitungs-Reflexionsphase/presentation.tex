\section{Überblick}

% ---- Folie 3: Kurzueberblick ueber das Datenbankdesign ----
\begin{frame}{Überblick über das Datenbankdesign}
  \small
  \begin{itemize}
    \item \textbf{Verwendetes DBMS:} SQLite 3 -- serverlos, eine einzelne Datei, voll SQL-faehig
      (CHECK, FOREIGN KEY, TRIGGER, JOIN); laut Aufgabenstellung frei waehlbar.
    \item \textbf{Anzahl Tabellen:} 17 (13 Entitaeten + 2 Dreifachbeziehungstabellen
      \texttt{ausleihe}/\texttt{verfuegbarkeit} + 2 M:N-Tabellen \texttt{buch\_autor}/\texttt{buch\_genre}).
    \item \textbf{Normalform:} 3. Normalform (3NF) -- keine transitiven Abhaengigkeiten,
      Autor/Verlag/Genre/Sprache konsequent ausgelagert.
    \item \textbf{Gegenueber Phase 1 umgesetztes Tutor-Feedback:}
      \begin{itemize}
        \footnotesize
        \item BUCH (Werk) und BUCHEXEMPLAR (physische Kopie inkl. Eigentuemer:in) getrennt.
        \item BEWERTUNG per eindeutigem FK an \texttt{ausleihe\_id} gekoppelt statt eigener Dreifachbeziehung.
        \item Durchgaengige CHECK-Constraints (Sterne 1--5, Versandkosten $\geq$ 0, \texttt{startzeit < endzeit}).
        \item Negativtestfaelle fuer alle zentralen Geschaeftsregeln (siehe Folien 20--22).
      \end{itemize}
  \end{itemize}
\end{frame}

\section{Constraints und Geschäftsregeln}

\begin{frame}[fragile]{Trigger fuer Geschaeftsregeln (nicht per CHECK abbildbar)}
  SQLite erlaubt keine Subqueries in \texttt{CHECK}-Constraints -- Regeln, die auf andere
  Tabellen verweisen, werden daher per \texttt{TRIGGER} durchgesetzt:
  \begin{sqlbox}
CREATE TRIGGER trg_ausleihe_kein_eigenausleih
BEFORE INSERT ON ausleihe FOR EACH ROW
WHEN NEW.nutzer_id = (SELECT eigentuemer_id FROM buchexemplar
                       WHERE exemplar_id = NEW.exemplar_id)
BEGIN SELECT RAISE(ABORT, 'Nutzer kann eigenes Buchexemplar nicht ausleihen'); END;

CREATE TRIGGER trg_bewertung_nur_nach_rueckgabe
BEFORE INSERT ON bewertung FOR EACH ROW
WHEN (SELECT status FROM ausleihe WHERE ausleihe_id = NEW.ausleihe_id) != 'zurueckgegeben'
BEGIN SELECT RAISE(ABORT, 'Bewertung nur fuer abgeschlossene Ausleihen erlaubt'); END;
  \end{sqlbox}
\end{frame}

\section{Zusammenfassung und Reflexion}

\begin{frame}{Zusammenfassung der Implementierung}
  \begin{itemize}
    \item Das ER-Modell aus Phase 1 (13 Entitaeten, 2 Dreifachbeziehungen) wurde vollstaendig
      in SQLite als 17 physische Tabellen umgesetzt: 13 Entitaetstabellen, 2 Dreifachbeziehungs-
      tabellen (\texttt{ausleihe}, \texttt{verfuegbarkeit}) und 2 M:N-Verknuepfungstabellen
      (\texttt{buch\_autor}, \texttt{buch\_genre}).
    \item Alle 17 Tabellen enthalten mindestens 10 realistische Dummy-Datensaetze
      (insgesamt $>$ 190 Zeilen), erzeugt und verifiziert per Python-\texttt{sqlite3}-Skript.
    \item Datenintegritaet wird durch NOT NULL, UNIQUE, Fremdschluessel (inkl. \texttt{ON
      DELETE CASCADE} wo fachlich sinnvoll) sowie CHECK-Constraints fuer alle relevanten
      Wertebereiche sichergestellt; zwei Trigger decken Geschaeftsregeln ab, die CHECK allein
      nicht abbilden kann.
    \item Fuer jede Entitaet liegt mindestens ein erfolgreicher Testfall vor; zusaetzlich
      wurden 6 gezielt ungueltige Faelle formuliert und ausgefuehrt, die alle korrekt
      abgelehnt wurden -- direkte Umsetzung des Tutor-Feedbacks aus Phase 1.
  \end{itemize}
\end{frame}

\begin{frame}{Herausforderungen und Erkenntnisse}
  \small
  \begin{itemize}
    \item \textbf{Werk vs. Exemplar:} Erst durch das Tutor-Feedback wurde klar, dass
      \texttt{eigentuemer\_id} nicht auf BUCH gehoert, sondern auf ein eigenes
      BUCHEXEMPLAR -- sonst koennte ein Buchtitel nur einer Person gehoeren.
    \item \textbf{Bewertung ohne Ausleihe verhindern:} Eine reine Dreifachbeziehung
      Bewertender--Buch--Eigentuemer haette nicht ausgeschlossen, dass ohne echten
      Ausleihvorgang bewertet wird. Die Kopplung ueber \texttt{ausleihe\_id} (UNIQUE)
      loest das strukturell, nicht nur per Anwendungslogik.
    \item \textbf{SQLite-Grenzen:} Keine nativen ENUM-Typen und keine Subqueries in
      CHECK-Constraints -- das erzwang saubere CHECK-Listen bzw. Trigger, was die
      Constraints expliziter macht als ein einfacher ENUM-Typ.
    \item \textbf{Naechster Schritt (Phase 3):} Weitere Indizes fuer Abfrageperformance
      sowie Ausbau der Umkreissuche (Haversine-Berechnung auf Basis von \texttt{standort}).
  \end{itemize}
\end{frame}
